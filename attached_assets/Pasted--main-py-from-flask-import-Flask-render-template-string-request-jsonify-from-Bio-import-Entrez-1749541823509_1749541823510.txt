# Mapeo científico fijo de bases a atributos visuales
BASE_ART_MAP = {
    'A': {'color': '#FF5555', 'size': 10, 'symbol': 'circle'},
    'T': {'color': '#FFFF50', 'size': 12, 'symbol': 'diamond'},
    'C': {'color': '#5555FF', 'size': 8, 'symbol': 'square'},
    'G': {'color': '#55FF55', 'size': 14, 'symbol': 'star'},
    'N': {'color': '#AAAAAA', 'size': 6, 'symbol': 'x'}
}

def obtener_secuencia(organismo):
    """Obtiene secuencia de ADN desde NCBI"""
    try:
        handle = Entrez.esearch(db="nucleotide", term=organismo, retmax=1)
        record = Entrez.read(handle)
        handle.close()
        
        if not record["IdList"]:
            return None
            
        seq_id = record["IdList"][0]
        handle = Entrez.efetch(db="nucleotide", id=seq_id, rettype="fasta", retmode="text")
        seq_record = SeqIO.read(handle, "fasta")
        handle.close()
        
        return seq_record
        
    except Exception as e:
        print(f"Error NCBI: {str(e)}")
        return None

def generar_visualizacion(seq_record):
    """Genera arte científico del ADN"""
    secuencia = str(seq_record.seq).upper()[:500]  # Limita a 500 bases
    
    # Datos para el gráfico
    posiciones = []
    bases = []
    propiedades = []
    
    for i, base in enumerate(secuencia):
        base = base if base in BASE_ART_MAP else 'N'
        posiciones.append(i)
        bases.append(base)
        propiedades.append(BASE_ART_MAP[base])
    
    # Crear figura Plotly
    fig = go.Figure()
    
    for base in set(bases):
        indices = [i for i, b in enumerate(bases) if b == base]
        fig.add_trace(go.Scatter(
            x=[posiciones[i] for i in indices],
            y=[1]*len(indices),
            mode='markers',
            marker=dict(
                color=BASE_ART_MAP[base]['color'],
                size=BASE_ART_MAP[base]['size'],
                symbol=BASE_ART_MAP[base]['symbol']
            ),
            name=f"Base {base}"
        ))
    
    fig.update_layout(
        title=f"ADN de {seq_record.id}",
        xaxis_title="Posición en secuencia",
        yaxis_visible=False,
        showlegend=True,
        plot_bgcolor='rgba(0,0,0,0)'
    )
    
    # Convertir imagen a base64
    img_bytes = fig.to_image(format="png")
    img_b64 = base64.b64encode(img_bytes).decode('utf-8')
    
    return img_b64, GC_content(secuencia), seq_record.description

@app.route("/", methods=["GET", "POST"])
def home():
    organismo = ""
    resultados = None
    error = ""
    
    if request.method == "POST":
        organismo = request.form.get("organismo", "").strip()
        if organismo:
            seq_record = obtener_secuencia(organismo)
            if seq_record:
                img_b64, gc, desc = generar_visualizacion(seq_record)
                resultados = {
                    "imagen": img_b64,
                    "gc": f"{gc:.2f}",
                    "longitud": len(seq_record.seq),
                    "id": seq_record.id,
                    "descripcion": desc
                }
            else:
                error = f"No se encontraron resultados para: {organismo}"
    
    return render_template_string(HTML_TEMPLATE, 
                               organismo=organismo,
                               resultados=resultados,
                               error=error)

@app.route("/descargar")
def descargar():
    organismo = request.args.get("organismo", "ADN")
    seq_record = obtener_secuencia(organismo)
    if seq_record:
        img_b64, _, _ = generar_visualizacion(seq_record)
        img_bytes = base64.b64decode(img_b64)
        return img_bytes, 200, {
            'Content-Type': 'image/png',
            'Content-Disposition': f'attachment; filename=dna_art_{organismo.replace(" ", "_")}.png'
        }
    return "Organismo no encontrado", 404
